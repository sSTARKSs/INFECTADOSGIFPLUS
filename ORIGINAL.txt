function genTxt(t) {
    if (!t.txt) return '';
    if (t.tipo === "3l") return '<div class="nombre-flotante texto-tres-lineas"><span>' + t.txt + '</span></div>';
    if (t.tipo === "2l") return '<div class="nombre-flotante texto-dos-lineas"><span>' + t.txt + '</span></div>';

}

function rotar() {

        elementos[bloqueActual].classList.add('activo');

}
